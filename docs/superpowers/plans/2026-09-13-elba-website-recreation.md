# Elba.no Recreation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild elba.no (a Framer site) as a static Next.js site with identical text, images, video, document and structure, plus a working Resend-backed contact form.

**Architecture:** The repo already contains the skralli.is recreation's pipeline and shared components (bootstrap commit `e21130a`). A Playwright scraper captures every live route into per-page JSON (ordered content blocks), downloads every image, the hero video and the PDF, and takes reference screenshots. Pages are then built from that JSON with the shared component set; product and article pages are one template each driven by typed data in `lib/` generated from the scrape. A verification script diffs visible text and image counts against the live site per route.

**Tech Stack:** Next.js 16 (App Router), React 19, TypeScript, Tailwind CSS 4 (`@theme` in `app/globals.css`), `next/font/google`, `next/image`, `resend`, Playwright (dev), Vitest (dev). Node 22. npm.

**Spec:** `docs/superpowers/specs/2026-09-13-elba-website-recreation-design.md`

## Global Constraints

- Text is copied **verbatim** from the live site (Norwegian Bokmål), including punctuation, capitalization and typos. Never "improve" copy. The only deviations are the four listed in the spec: kontakt-oss button `Senda!` → `Send`; footer "Artikler" link → `/artikler`; article `<title>` suffix `— Skralli` → `— Elba`; U+200B slug redirects and the omitted table-widget chrome.
- Every image visible on a live page must exist locally under `public/` and be rendered on the equivalent new page. No hotlinking to `framerusercontent.com`. Same for the hero video and the PDF.
- Fonts: Figtree and Inter through `next/font/google` (already in `app/fonts.ts`). No `<link>` to Google Fonts. Satoshi stays only if Task 11's measurement shows the live stat counters use it.

- Secrets (`RESEND_API_KEY`, `EMAIL_FROM`, `CONTACT_TO`) live only in `.env.local` (gitignored) and Vercel. Never in code or commits.

async function downloadAsset(url, kind) {
  const clean = stripParams(url);
  if (seenImages.has(clean)) return seenImages.get(clean);
  const dir = kind === "video" ? "video" : "docs";
  const name = kind === "video" ? "hero.mp4" : path.basename(new URL(clean).pathname);
  let result;
  try {
    const res = await fetch(clean, { headers: { "User-Agent": "Mozilla/5.0" } });
    if (!res.ok) throw new Error(`download ${clean} -> ${res.status}`);
    const local = `/${dir}/${name}`;
    await mkdir(path.join("public", dir), { recursive: true });
    await writeFile(path.join("public", local), Buffer.from(await res.arrayBuffer()));
    result = { local };
  } catch (e) {
    result = { local: null, error: String(e.message || e) };
    console.error(`  ! asset failed: ${clean} — ${result.error}`);
  }
  seenImages.set(clean, result);
  return result;
}

// Reads every <table> AFTER Framer hydrates the page. The server HTML of
// elba.no's product pages only contains the Table widget's demo rows
// (Name / Email / Role / Status); the real spec rows are rendered
// client-side, paginated ("Page 1 of N") and duplicated once for the
// desktop and once for the mobile layout. This script clicks through the
// pager, then drops exact-duplicate tables.
import { chromium } from "playwright";
import { mkdir, writeFile } from "node:fs/promises";
import { ROUTES, LIVE, livePath } from "./routes.mjs";

const PLACEHOLDER = JSON.stringify(["Name", "Email", "Role", "Status"]);

async function readTables(page) {
  return page.evaluate(() => {
    const clean = (s) => s.replace(/\s+/g, " ").trim();
    const visible = (el) => { const r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; };
    const out = [];
    for (const table of document.querySelectorAll("table")) {
      if (!visible(table)) continue;
      const headers = [...table.querySelectorAll("thead th, thead td")].map((c) => clean(c.textContent));
      const rows = [...table.querySelectorAll("tbody tr")].map((tr) => [...tr.querySelectorAll("td, th")].map((c) => clean(c.textContent)));
      let heading = null;
      let node = table;
      while (node && !heading) {
        let sib = node.previousElementSibling;
        while (sib && !heading) { const h = sib.matches("h1,h2,h3,h4,h5,h6") ? sib : sib.querySelector("h1,h2,h3,h4,h5,h6"); if (h) heading = clean(h.textContent); sib = sib.previousElementSibling; }
        node = node.parentElement;
      }
      out.push({ headers, rows, precedingHeading: heading });
    }
    return out;
  });
}

  // Collect page 1 of every widget, then click every enabled "Next" until none is enabled.
  const pages = [await readTables(page)];
  for (let i = 0; i < 20; i++) {
    const next = page.locator("button:has-text('Next'):not([disabled]), [role=button]:has-text('Next'):not([aria-disabled='true'])");
    if ((await next.count()) === 0) break;
    await next.first().click();
    await page.waitForTimeout(600);
    pages.push(await readTables(page));
  }
  // Merge by DOM position: table i of every snapshot is the same widget
  // (clicking "Next" never adds or removes tables), so later snapshots only
  // contribute rows not seen yet. Never merge by heading — rørender repeats
  // the same variant heading over several distinct tables.
  const base = pages[0].map((t) => ({ ...t, rows: [...t.rows] }));
  for (const snapshot of pages.slice(1)) {
    if (snapshot.length !== base.length) { console.warn(`  ! table count changed between pager clicks (${base.length} → ${snapshot.length}); keeping page 1 only`); break; }
    snapshot.forEach((t, i) => { for (const row of t.rows) if (!base[i].rows.some((r) => JSON.stringify(r) === JSON.stringify(row))) base[i].rows.push(row); });
  }
  // Drop the widget's demo table and the exact desktop/mobile duplicates.
  const seen = new Set();
  const out = [];
  for (const t of base) {
    if (JSON.stringify(t.headers) === PLACEHOLDER) continue;
    const key = JSON.stringify([t.precedingHeading, t.headers, t.rows]);
    if (seen.has(key)) continue;
    seen.add(key);
    out.push(t);
  }
  await page.context().close();
  return out.map((t, index) => ({ index, ...t }));
}

  if (p.website) return "Sending mislyktes.";
  if (!p.navn?.trim()) return "Vennligst fyll inn navn.";

  if (!p.melding?.trim()) return "Vennligst skriv en melding.";
  return null;

const ESCAPE_MAP: Record<string, string> = { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" };
const esc = (s: string) => s.replace(/[&<>"']/g, (c) => ESCAPE_MAP[c]!);

export function buildContactEmail(p: ContactPayload): { subject: string; text: string; html: string } {

  const text = [`Navn: ${p.navn}`, `E-post: ${p.epost}`, `Selskap: ${p.selskap || "-"}`, "", "Melding:", p.melding].join("\n");

  return { subject, text, html };
}

// Generates lib/produkter.ts from docs/scrape/produkter__*.json (blocks:
// hero image, H1, breadcrumb, intro text, then per variant an optional
// H3 + up to two images) and docs/scrape/tables.json (hydrated tables,
// keyed by precedingHeading). Run: npm run gen-produkter
import { readFileSync, readdirSync, writeFileSync } from "node:fs";
import { join, dirname } from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = join(dirname(fileURLToPath(import.meta.url)), "..");
const SCRAPE = join(ROOT, "docs/scrape");
const tablesByRoute = Object.fromEntries(JSON.parse(readFileSync(join(SCRAPE, "tables.json"), "utf8")).routes.map((r) => [r.route, r.tables]));
const CTA = "Send oss en forespørsel";
const img = (b) => ({ src: b.local, alt: b.alt || "", width: b.width, height: b.height });
const strip = (s) => s.replace(/​/g, "");

const products = [];
for (const file of readdirSync(SCRAPE).filter((f) => f.startsWith("produkter__") && f.endsWith(".json")).sort()) {
  const j = JSON.parse(readFileSync(join(SCRAPE, file), "utf8"));
  const id = strip(j.route.slice("/produkter/".length));
  const blocks = j.blocks.filter((b) => !(b.type === "image" && b.role === "logo"));
  const h1 = blocks.findIndex((b) => b.type === "heading" && b.level === 1);
  const hero = blocks.slice(0, h1).find((b) => b.type === "image");
  const title = strip(blocks[h1].text);
  // body = after breadcrumb (first block after h1 that is not text/link "Produkter"/"›"/">" crumbs)
  let i = h1 + 1;
  while (i < blocks.length && (blocks[i].type === "link" || (blocks[i].type === "text" && /^(>|›|Hjem|Produkter)$/.test(blocks[i].text.trim())) || (blocks[i].type === "text" && strip(blocks[i].text) === title))) i++;
  const body = blocks.slice(i).filter((b) => !(b.type === "heading" && b.text === CTA) && !(b.type === "link" && b.text === "Kontakt oss"));
  const intro = [];
  const tables = [];
  const live = tablesByRoute[j.route] ?? [];
  let pendingImages = [];
  let pendingHeading = null;
  const seenTables = new Set();
  const flushTable = (heading) => {
    const t = live.find((t) => !seenTables.has(t.index) && (t.precedingHeading ?? null) === (heading ?? null)) ?? live.find((t) => !seenTables.has(t.index));
    if (!t) return;
    seenTables.add(t.index);
    tables.push({ heading, headers: t.headers, rows: t.rows, images: pendingImages });
    pendingImages = [];
  };
  for (const b of body) {
    if (b.type === "heading" && b.level === 3) { if (pendingHeading !== null || pendingImages.length) flushTable(pendingHeading); pendingHeading = b.text; continue; }
    if (b.type === "image") { pendingImages.push(img(b)); continue; }
    if (b.type === "text" && tables.length === 0 && pendingHeading === null && !live.some((t) => t.rows.flat().includes(b.text) || t.headers.includes(b.text))) { intro.push(b.text); continue; }
  }
  if (pendingHeading !== null || pendingImages.length || tables.length < live.length) flushTable(pendingHeading);
  // Any hydrated table not yet matched (page with no H3s) is appended in order.
  for (const t of live) if (!seenTables.has(t.index)) { seenTables.add(t.index); tables.push({ heading: t.precedingHeading === title ? null : t.precedingHeading, headers: t.headers, rows: t.rows, images: [] }); }

  const dedup = []; const keys = new Set();
  for (const t of tables) { const k = JSON.stringify([t.heading, t.headers, t.rows]); if (!keys.has(k)) { keys.add(k); dedup.push(t); } }
